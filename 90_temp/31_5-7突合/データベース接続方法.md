#### 関数化した接続方法（推奨）
```python
import cx_Oracle
import pandas as pd

def ej_data_get(sql):
    """
    EJシステム（Oracle Database）に接続してSQLを実行する

    Args:
        sql (str): 実行するSQL文

    Returns:
        pd.DataFrame: クエリ結果のDataFrame
    """
    try:
        # EJシステム接続情報
        host = '172.17.107.102'
        port = '1521'
        service_name = 'EXPJ'
        username = 'EXPJ2'
        password = 'EXPJ2'

        # 接続文字列
        connection_string = f"{username}/{password}@{host}:{port}/{service_name}"

        # データベース接続
        connection = cx_Oracle.connect(connection_string)

        # SQLを実行してDataFrameに変換
        df = pd.read_sql(sql, connection)

        # 接続を閉じる
        connection.close()

        return df

    except Exception as e:
        print(f"EJシステムへの接続でエラーが発生しました: {str(e)}")
        raise

# 使用例
sql = """
SELECT ITEM_CD, PUCH_FIXED_LT
FROM EXPJ2.M_ITEM
WHERE ITEM_CD = 'K-24913AB'
"""

df = ej_data_get(sql)
print(df)
```

### 主要テーブル一覧

| テーブル名 | 説明 | 主要カラム |
|-----------|------|-----------|
| M_ITEM | 品目マスタ | ITEM_CD, PUCH_FIXED_LT, PRODUCT_TYP |
| M_PUCH_UNIT_COST_H | 仕入単価ヘッダ | ITEM_CD, VEND_CD, PUCH_PRIORITY_REF_NO |
| M_PUCH_UNIT_COST | 仕入単価 | ITEM_CD, VEND_CD, UNIT_COST, EFF_PHASE_IN_DATE, PUCH_SIZE |
| T_RLSD_PUCH_ODR | 発注履歴 | ITEM_CD, VEND_CD, UNIT_COST, PUCH_ODR_DLV_DATE |

### Oracle IN句の制限対応
Oracleでは IN 句に1000件を超える値を指定できません。バッチ処理で対応します。

```python
def batch_query_ej(item_list, batch_size=900):
    """
    大量のITEM_CDを900件ずつバッチ処理でクエリする

    Args:
        item_list (list): ITEM_CDのリスト
        batch_size (int): バッチサイズ（デフォルト900）

    Returns:
        pd.DataFrame: 全バッチの結果を結合したDataFrame
    """
    all_results = []

    for i in range(0, len(item_list), batch_size):
        batch = item_list[i:i + batch_size]
        item_sql_list = "','".join(batch)

        sql = f"""
        SELECT ITEM_CD, PUCH_FIXED_LT
        FROM EXPJ2.M_ITEM
        WHERE ITEM_CD IN ('{item_sql_list}')
        """

        batch_df = ej_data_get(sql)
        all_results.append(batch_df)
        print(f"バッチ{i//batch_size + 1}: {len(batch)}件 -> {len(batch_df)}行取得")

    # 全バッチを結合
    if all_results:
        result_df = pd.concat(all_results, ignore_index=True)
    else:
        result_df = pd.DataFrame()

    return result_df
```

### 接続サンプルコード

#### 基本的な接続方法

```python
cursor = connection.cursor()

# SQL実行例
cursor.execute("SELECT COUNT(*) FROM DATA_RES_CAPA")

print(f"DATA_RES_CAPA件数: {result[0]}")

# 接続を閉じる
cursor.close()
connection.close()
```

#### SQLAlchemyを使用した接続方法（推奨）
```python
from sqlalchemy import create_engine
import pandas as pd

def krd_data_get(sql):
    """
    KRDシステム（MySQL Database）に接続してSQLを実行する

    Args:
        sql (str): 実行するSQL文

    Returns:
        pd.DataFrame: クエリ結果のDataFrame
    """
    try:
        # KRDシステム接続情報
        host = 'krd'
        user = 'pfw'
        password = 'mejiriHoo'
        database = 'machin'

        # 接続文字列（SQLAlchemy形式）
        connection_string = f"mysql+pymysql://{user}:{password}@{host}/{database}"

        # エンジン作成
        engine = create_engine(connection_string)

        # SQLを実行してDataFrameに変換
        df = pd.read_sql(sql, engine)

        # エンジンを閉じる
        engine.dispose()

        return df

    except Exception as e:
        print(f"KRDシステムへの接続でエラーが発生しました: {str(e)}")
        raise

# 使用例
sql = """
SELECT 品番, 工程, 設備, 段取時間, 加工時間
FROM DATA_RES_CAPA
WHERE 品番 LIKE '5156-%'
LIMIT 10
"""

df = krd_data_get(sql)
print(df)
```

### 主要テーブル一覧

| テーブル名 | 説明 | 主要カラム |
|-----------|------|-----------|
| DATA_RES_CAPA | 加工能力データ | 品番, 工程, 設備, 段取時間, 加工時間 |

### 文字コード対応
KRDデータベースは日本語を含むため、文字コード指定が重要です。

```python

connection = pymysql.connect(
    host='krd',

    database='machin',
    charset='utf8mb4',  # 日本語対応
    cursorclass=pymysql.cursors.DictCursor  # 辞書形式で結果取得
)
```

## 3. 両方のデータベースを使用する例

### EJとKRDのデータを突合する処理例
```python
import cx_Oracle
import pandas as pd
from sqlalchemy import create_engine

def ej_data_get(sql):
    """EJデータベースからデータ取得"""
    connection_string = "EXPJ2/EXPJ2@172.17.107.102:1521/EXPJ"
    connection = cx_Oracle.connect(connection_string)
    df = pd.read_sql(sql, connection)
    connection.close()
    return df

def krd_data_get(sql):
    """KRDデータベースからデータ取得"""
    connection_string = "mysql+pymysql://pfw:mejiriHoo@krd/machin"
    engine = create_engine(connection_string)
    df = pd.read_sql(sql, engine)
    engine.dispose()
    return df

# EJからM_ITEMデータを取得
ej_sql = """
SELECT ITEM_CD, PUCH_FIXED_LT, PRODUCT_TYP
FROM EXPJ2.M_ITEM
WHERE PRODUCT_TYP IN (6, 7, 8)
"""
ej_df = ej_data_get(ej_sql)
print(f"EJデータ取得: {len(ej_df)}件")

# KRDからDATA_RES_CAPAデータを取得
krd_sql = """
SELECT DISTINCT 品番
FROM DATA_RES_CAPA
"""
krd_df = krd_data_get(krd_sql)
print(f"KRDデータ取得: {len(krd_df)}件")

# 品番で突合
# EJのITEM_CDとKRDの品番をマージ
merged_df = pd.merge(
    ej_df,
    krd_df,
    left_on='ITEM_CD',
    right_on='品番',
    how='inner'
)

print(f"突合結果: {len(merged_df)}件")
print(merged_df.head())

# 結果をCSVに出力
merged_df.to_csv('突合結果.csv', encoding='utf-8-sig', index=False)
```

### Oracle接続エラー
**エラー**: `DPI-1047: Cannot locate a 64-bit Oracle Client library`

**解決策**: Oracle Instant Client 19.23をインストール
```bash
# ダウンロード: https://www.oracle.com/database/technologies/instant-client/downloads.html
# 環境変数PATHにOracle Instant Clientのパスを追加
```

### MySQL接続エラー
**エラー**: `Can't connect to MySQL server on 'krd'`

**解決策**:
1. ホスト名'krd'が名前解決できるか確認
2. ネットワーク接続を確認
3. MySQLサーバーが起動しているか確認

### 文字化け問題
**EJ（Oracle）**: クエリ結果が文字化けする場合
```python

```


## 注意事項

1. **接続情報の取り扱い**: パスワード等の機密情報は環境変数や設定ファイルで管理することを推奨
2. **接続のクローズ**: 必ず `connection.close()` または `engine.dispose()` で接続を閉じる
3. **バッチ処理**: 大量データ処理時はメモリ使用量に注意
4. **タイムアウト設定**: 長時間のクエリにはタイムアウト設定を検討
5. **トランザクション**: 更新系処理の場合は適切にトランザクション管理を行う
